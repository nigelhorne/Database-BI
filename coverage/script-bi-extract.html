1#!/usr/bin/env perl
2# bi-extract -- build the extracts of accelerated datasets (Database::BI, Phase 7.5)
3#
4# An accelerated dataset is read from a SQLite copy of itself, an extract, for
5# as long as the extract still matches the file it was built from.  This script
6# marks a dataset accelerated and builds its extract, rebuilds every accelerated
7# dataset's extract, or stops accelerating one.  It prints what it did.
8#
9# An extract is a cache under <model_dir>/extracts/.  Deleting that directory
10# loses nothing: the application reads each file again until this is rerun.
11# Nothing is ever written under data_dir.
12#
13# Usage:
14#   bi-extract --spec <l-spec> [--spec <l-spec> ...]
15#   bi-extract --all
16#   bi-extract --remove --spec <l-spec>
17#   bi-extract --list
18#   options: [--data-dir <dir>] [--model-dir <dir>]
19#
20# Exit codes:  0 = ok
21#              1 = an extract could not be built (the others still are)
22#              2 = bad arguments
23#
24# Suitable for cron, after the data files are refreshed:
25#   15 2 * * * cd /opt/myapp && ./script/bi-extract --all
