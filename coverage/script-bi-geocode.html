1#!/usr/bin/env perl
2# bi-geocode -- geocode-cache warming CLI for Database::BI
3#
4# Opens a dataset spec, geocodes every distinct value of the named column
5# (or of several, --col repeated, joined with ", ": an address, then its country),
6# with --text added to the end of each place ("Virginia, USA"),
7# and writes the resolved coordinates into the shared geocode cache in
8# models.json.  Subsequent map views for the same dataset serve from the
9# cache without making any geocoder requests.
10#
11# Places are looked up in Geo::Coder::Free's local data first, then on
12# OpenStreetMap (Nominatim) at one a second, with no limit on how many: the
13# map page asks Nominatim for at most map_geocode_online_max places, and this
14# script is how the rest are found.  --offline keeps every lookup local.
15#
16# The script is idempotent: a second run against the same column with a
17# warm cache resolves zero additional places (and says so).
18#
19# Usage:
20#   bi-geocode --spec <l-spec> --col <column> [--col <column> ...]
21#              [--text <text>] [--data-dir <dir>] [--model-dir <dir>]
22#              [--openaddr <path>] [--region <CC>]
23#              [--offline] [--online-max <n>]
24#              [--dry-run] [--verbose]
25#
26# Exit codes:  0 = ok
27#              1 = column not found or no records loaded
28#              2 = bad arguments / data source unavailable
29#
30# Suitable for cron:
31#   0 2 * * * cd /opt/myapp && ./script/bi-geocode \
32#       --spec table:customers --col city --region US --verbose
